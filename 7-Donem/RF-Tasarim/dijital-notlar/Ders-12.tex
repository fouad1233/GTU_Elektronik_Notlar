\item Simple design process
\item Mixing spurs are reduced
\end{itemize}

\begin{ornek}
\textbf{FSK Direct Conversion Receiver}
\[ x_{FSK}(t) = A_0\cos[(\omega_c + a\omega_1)t],\quad a = \pm 1 \]
\elcizimi[0.5]{sekiller/Ders-12-s1-2.png}[1]
\elcizimi[0.65]{sekiller/Ders-12-s1-3.png}[1]
\begin{align*}
x_{BB,I} &= A_0\cos[(\omega_c + a\omega_1)t]\cdot B\cos\omega_0 t\\
&= \overset{\scriptstyle A_1}{\frac{A_0 B}{2}}\bigl[\cos(2\omega_c + a\omega_1)t + \cos a\omega_1 t\bigr] \;\xrightarrow{\text{LPF}}\; x_{BB,I} = A_1\cos a\omega_1 t\\
x_{BB,Q} &= A_0\cos[(\omega_c + a\omega_1)t]\cdot B\sin\omega_0 t\\
&= \frac{A_0 B}{2}\bigl[\sin(2\omega_c + a\omega_1)t - \sin a\omega_1 t\bigr]\\
&\hookrightarrow\; x_{BB,Q} = -A_1\sin a\omega_1 t
\end{align*}
\end{ornek}

\subsection{(4) Flicker noise}
\elcizimi[0.7]{sekiller/Ders-12-s5-1.png}[5]
\[
\begin{array}{@{}l@{\quad}l@{}}
f_{BW} & : \text{Half of RF channel Band width}\\
S_{1/f}(f) = \dfrac{a}{f} & : \text{PSD of flicker noise}\\
S_{th}(f) = S_{th} & : \text{PSD of thermal noise}\\
f_c & : \text{corner frequency}
\end{array}
\]

We ignore noise effect below $\dfrac{f_{BW}}{1000}$ since they vary slowly

\[ S_{1/f}(f) = \frac{a}{f} \;\rightarrow\; S_{1/f}(f_c) = \frac{a}{f_c} = S_{th} \;\rightarrow\; a = f_c S_{th} \]
\[ P_{n1} = \int_{\frac{f_{BW}}{1000}}^{f_c} \frac{a}{f}\,df + \int_{f_c}^{f_{BW}} S_{th}\,df \]
\[ = \underset{f_c S_{th}}{a}\,\ln\frac{1000 f_c}{f_{BW}} + (f_{BW} - f_c)S_{th} \]
\formulkutu[önemli]{P_{n1} = \left(5{,}9 + \ln\frac{f_c}{f_{BW}}\right) f_c S_{th} + f_{BW}\cdot S_{th}}

\[ \text{In the absence of flicker noise} = \kutula{P_{n2} \approx S_{th}\cdot f_{BW}} \]

\subsubsection{Flicker noise Penalty}
\hfill $\downarrow\downarrow$ azaltmak istiyoruz
\formulkutu[önemli]{\frac{P_{n1}}{P_{n2}} = 1 + \left(5{,}9 + \ln\frac{f_c}{f_{BW}}\right)\cdot\frac{f_c}{f_{BW}}}

Phase \& gain Error

\sayfa{7}
\elcizimi[0.65]{sekiller/Ders-12-s7-1.png}[7]
\[ \varepsilon \rightarrow \text{gain error} \qquad \theta \rightarrow \text{phase error} \]

\[ x_{in}(t) = a\cos\omega_c t + b\sin\omega_c t \;\rightarrow\; \text{QPSK signal} \]
\begin{align*}
x_{LO,I}(t) &= 2\left(1 + \tfrac{\varepsilon}{2}\right)\cos\left(\omega_c t + \tfrac{\theta}{2}\right)\\
x_{LO,Q}(t) &= 2\left(1 - \tfrac{\varepsilon}{2}\right)\sin\left(\omega_c t - \tfrac{\theta}{2}\right)
\end{align*}
When $\varepsilon = 0,\ \theta = 0$
\[ \rightarrow\ x_{BB,I}(t) = a,\qquad x_{BB,Q}(t) = b \]

\subsubsection{Outputs}
\begin{align*}
x_{BB,I}(t) &= a\left(1 + \tfrac{\varepsilon}{2}\right)\cos\left(\tfrac{\theta}{2}\right) - b\left(1 + \tfrac{\varepsilon}{2}\right)\sin\left(\tfrac{\theta}{2}\right) \;\longrightarrow\\
x_{BB,Q}(t) &= -a\left(1 - \tfrac{\varepsilon}{2}\right)\sin\left(\tfrac{\theta}{2}\right) + b\left(1 - \tfrac{\varepsilon}{2}\right)\cos\left(\tfrac{\theta}{2}\right) \;\longrightarrow
\end{align*}

\subsubsection{Only gain mismatch ($\theta = 0$)}
\begin{align*}
x_{BB,I}(t) &= a\left(1 + \tfrac{\varepsilon}{2}\right)\\
x_{BB,Q}(t) &= b\left(1 - \tfrac{\varepsilon}{2}\right)
\end{align*}
\elcizimi[0.4]{sekiller/Ders-12-s7-2.png}[7]

\subsubsection{Only phase mismatch ($\varepsilon = 0$)}
\begin{align*}
x_{BB,I}(t) &= a\cos\left(\tfrac{\theta}{2}\right) - b\sin\left(\tfrac{\theta}{2}\right)\\
x_{BB,Q}(t) &= -a\sin\left(\tfrac{\theta}{2}\right) + b\cos\left(\tfrac{\theta}{2}\right)
\end{align*}
