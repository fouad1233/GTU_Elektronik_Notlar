\section{Receiver noise Floor}

\[ F = \frac{N_o + G\,kTB}{G}\cdot\frac{1}{kTB} \]

\[ \tikz[baseline=(sX.base)]{\node[inner sep=2pt](sX){$N_o = (F-1)\,kTB\cdot G$}; \draw[kirmizikalem,line width=0.8pt] (sX.south west) -- (sX.north east); \draw[kirmizikalem,line width=0.8pt] (sX.north west) -- (sX.south east);} \]

\[ F\,kTB = \frac{N_o}{G} + kTB = \text{noise Floor} \]

\[ \text{noise floor} = F\cdot kTB \]

$F \leftarrow$ Noise factor, \quad $k \leftarrow$ Boltzman constant, \quad $T \leftarrow$ temperature

\section{Sensitivity and Dynamic Range}

Sensitivity is the minimum\textsuperscript{input} signal which can be detected with the acceptable output SNR.

\elcizimi[0.45]{sekiller/Ders7-s3-2.png}[3]

\[ NF = \frac{(\mathrm{SNR})_{\mathrm{in}}}{(\mathrm{SNR})_{\mathrm{out}}} = \frac{P_{\mathrm{sig,in}}/P_{n,\mathrm{in}}}{\mathrm{SNR}_{\mathrm{out}}} \]

\[ P_{\mathrm{sig,in}} = \underbrace{P_{n,\mathrm{in}}}_{\substack{kTB\\ \text{if } R_s = R_{\mathrm{in}}}}\cdot F\cdot \mathrm{SNR}_{\mathrm{out}} \]

\[ P_{\mathrm{sig,in}} = \underbrace{kTB\cdot F}_{\substack{\text{noise}\\ \text{floor}}}\cdot \mathrm{SNR}_{\mathrm{out}} \]

\formulkutu{P_{\mathrm{sig,in(min)}}\Big|_{\mathrm{dB}} = \text{noise floor}_{\mathrm{dB}} + \mathrm{SNR}_{\mathrm{out,dB}}}

\sayfa{4}
\[ \underbrace{P_{\mathrm{sig,in(min)}}\Big|_{\mathrm{dB}}}_{\text{sensitivity}} = -174\,\frac{\mathrm{dBm}}{\mathrm{Hz}} + 10\log B + NF + \mathrm{SNR}_{\mathrm{out}}\Big|_{\mathrm{dB}} \]

\begin{ornek}
A GSM receiver has $\mathrm{SNR}_{\mathrm{out(min)}} = 12$ dB \\
channel $BW = 20$ khz \\
$NF = 9$ dB

\end{ornek}

\footnotetext{Claude notu: $10\log(20\cdot 10^3)\approx 43$ dB'dir; 53 dB, 200 kHz'lik bant genişliğine karşılık gelir (GSM kanalı 200 kHz).}
