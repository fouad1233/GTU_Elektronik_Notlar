What is the amount of rejection in BPF$_1$ at 5,2 GHz for SNR $=20$ dB \& IRR $=45$ dB
\end{ornek}

\begin{gather*}
\mathrm{SNR}\big|_{\mathrm{dB}}=S_{\mathrm{dBm}}-\overset{I_{\mathrm{dBm}}}{N_{\mathrm{dBm}}}\ge 20\,\mathrm{dB}\\
-85-N_{\mathrm{dBm}}\ge 20\,\mathrm{dB}\\
N_{\mathrm{dBm}}\le -105\,\mathrm{dBm}\\
S_{int}\big|_{\mathrm{dBm}}-|BPF|_{\mathrm{dB}}-|IRR|_{\mathrm{dB}}=N_{\mathrm{dBm}}\le -105\,\mathrm{dBm}\\
-10\,\mathrm{dBm}-|BPF_1|_{\mathrm{dB}}-45\le -105\,\mathrm{dBm}\\
|BPF_1|\ge 50\,\mathrm{dB}
\end{gather*}
