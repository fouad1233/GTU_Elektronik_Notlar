\[ \cos 2\pi f_0 t = \frac{e^{j2\pi f_0 t}}{2} + \frac{e^{-j2\pi f_0 t}}{2} \;\overset{F}{\longrightarrow}\; \frac12\bigl[\delta(f-f_0)+\delta(f+f_0)\bigr] \;\rightarrow\; \text{real} \]
\hfill $\downarrow$ 90$^\circ$ phase shift
\[ \sin 2\pi f_0 t = j\,\frac{e^{-j2\pi f_0 t}}{2} - j\,\frac{e^{j2\pi f_0 t}}{2} \;\overset{F}{\longrightarrow}\; \frac{j}{2}\bigl[-\delta(f-f_0)+\delta(f+f_0)\bigr] \;\rightarrow\; \text{Imag.} \]

\elcizimi[0.9]{sekiller/Ders-11-s3-3.png}[3]

\begin{align*}
\cos(\omega_c t - 90^\circ) &= \frac{e^{j(\omega_c t-90)}}{2} + \frac{e^{-j(\omega_c t-90)}}{2}\\
&= \frac{e^{j\omega_c t}}{2}\,e^{-j90} + \frac{e^{-j\omega_c t}}{2}\,e^{j90}
\end{align*}
\[ e^{-j90} = \cos 90 - j\sin 90 = -j \]
\[ e^{j90} = \cos 90 + j\sin 90 = j \]
\[ = -j\,\frac{e^{j\omega_c t}}{2} + j\,\frac{e^{-j\omega_c t}}{2} = \sin(\omega_c t) \]
