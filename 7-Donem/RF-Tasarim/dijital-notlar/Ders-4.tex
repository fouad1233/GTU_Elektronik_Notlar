\section{Instantaneous Power (anlık güç)}
\elcizimi[0.7]{sekiller/Ders-4-s2-1.png}[2]
\[ p(t) = v(t)\,i(t) = \frac{v^2(t)}{R} \]
Bizim için ortalama güç önemli
\[ P_{ave} = \frac{1}{T}\int_0^T p(t)\,dt = \frac{1}{R}\,\underline{\frac{1}{T}\int_0^T v^2(t)\,dt} \]
\[ \langle v^2(t)\rangle = \frac{1}{T}\int_0^T v^2(t)\,dt \qquad \text{$v^2(t)$'nin ortalaması} \]
\[ \sqrt{\langle v^2(t)\rangle} = V_{rms} \text{ (root mean square)} \]

\section{Complex Power}
\begin{align*}
V &= |V|e^{j\theta_V} & V_{rms} &= \frac{|V|}{\sqrt2}e^{j\theta_V}\\
I &= |I|e^{j\theta_I} & I_{rms} &= \frac{|I|}{\sqrt2}e^{j\theta_I}
\end{align*}
\[ S = P + jQ \]
\begin{align*}
S &= \frac12\left(V\cdot I^*\right) \text{ or } \left(V_{rms}\cdot I_{rms}^*\right)\\
&= \frac12\left(|V|e^{j\theta_V}\,|I|e^{-j\theta_I}\right) = \frac12|V||I|e^{j(\theta_V-\theta_I)}\\
&= \underbrace{\frac12|V||I|\cos(\theta_V-\theta_I)}_{P} + j\,\underbrace{\frac12|V||I|\sin(\theta_V-\theta_I)}_{Q}
\end{align*}

\sayfa{4}
\noindent P: average power or active Power \quad $P = \mathrm{Real}\{S\}$ \hfill [watt -- W]\\
Q: reactive power \quad $Q = \mathrm{Imag}\{S\}$ \hfill [VAR] Voltage Amper Reactive

\begin{center}
\renewcommand{\arraystretch}{1.8}
\begin{tabular}{l c c c}
 & \underline{Z} & \underline{P} & \underline{Q}\\
Resistor & \shortstack{$R$\\$\theta_V=\theta_I=0$} & \shortstack{$P=\dfrac{|V|^2}{R}>0$\\absorbs power} & $0$\\
Inductor & \shortstack{$j\omega L$\\$\theta_V-\theta_I=90^\circ$} & $0$ & \shortstack{$\dfrac{j|V|^2}{\omega L}>0$\\absorbs power (VAR)}\\
Capacitor & \shortstack{$\dfrac{1}{j\omega C}$\\$\theta_V-\theta_I=-90^\circ$} & $0$ & \shortstack{$-j|V|^2\omega C<0$\\generate VAR}\\
\end{tabular}
\end{center}
