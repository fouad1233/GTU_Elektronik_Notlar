\[ P(x|\theta) = \theta^x(1-\theta)^{1-x}, \quad \overset{\text{ya 0 ya 1}}{x\in\{0,1\}}, \quad \overset{\text{0 ile 1 arasında}}{\theta\in[0,1]} \]

\begin{align*}
L(\theta|\mathcal{X}) &= \prod_{i=1}^{N} p(x^i|\theta) \\
&= \prod_{i=1}^{N} \theta^{x_i}(1-\theta)^{(1-x_i)} \\
\ell(\theta|\mathcal{X}) &= \sum_{i=1}^{N}\Bigl[(x_i\log\theta)+(1-x_i)\log(1-\theta)\Bigr]
\end{align*}
\[ x_i \in \{0,1\} \;\downarrow\; \kutula{s=\sum_{i=1}^{N} x_i} \]

\[ \frac{\partial \ell(\theta|\mathcal{X})}{\partial\theta} = 0 = \frac{\partial(s\log\theta)}{\partial\theta} + \frac{\partial\bigl(\log(1-\theta)(N-s)\bigr)}{\partial\theta} \]
\[ = \frac{s}{\theta} - \frac{N-s}{1-\theta} \]
\[ \frac{N-s}{1-\theta} = \frac{s}{\theta} \]
\[ s - s\theta = N\theta - s\theta \]
\[ \hat\theta = \frac{s}{N} = \frac{\sum_{i=1}^{N} x_i}{N} \]

\[ p(x|\mu,\sigma^2) = \frac{1}{\sqrt{2\pi\sigma^2}}\,e^{-\frac{(x-\mu)^2}{2\sigma^2}} \]

\elcizimi[0.4]{sekiller/Ders-4-s3-2.png}[3]

\begin{align*}
L(\theta|\mathcal{X}) &= \prod_{i=1}^{N} p(x_i|\theta) \\
&= \Bigl(\frac{1}{\sqrt{2\pi\sigma^2}}\Bigr)^{N}\prod_{i=1}^{N} e^{-\frac{(x-\mu)^2}{2\sigma^2}} \\
\ell(\theta|\mathcal{X}) &= -N\log(2\pi\sigma^2) - \sum_{i=1}^{N}\frac{(x-\mu)^2}{2\sigma^2}
\end{align*}
\[ \frac{\partial\,\ell(\theta|\mathcal{X})}{\partial\mu} = \frac{1}{2\sigma^2}\sum_{i=1}^{N} 2(x_i-\mu) \]
\[ 0 = \frac{\sum_{i=1}^{N} x_i - N\mu}{\sigma^2} \]

\begin{align*}
E\{m\} &= E\left\{\frac{\sum_{t=1}^{N} x_t}{N}\right\} = \frac{1}{N}\sum_{t=1}^{N} E\{x^t\} \\
&= \frac{1}{N}\sum_{t=1}^{N}\mu = \frac{N\cdot\mu}{N} = \mu
\end{align*}

\begin{align*}
\mathrm{Var}\{m\} &= \mathrm{Var}\left\{\frac{\sum_{t=1}^{N} x_t}{N}\right\} = \frac{1}{N^2}\sum_{t=1}^{N}\mathrm{Var}\{x_t\} = \frac{N\cdot\mathrm{Var}\{x_t\}}{N^2} \\
&= \frac{\mathrm{Var}[x_t]}{N} = \frac{\sigma^2}{N}
\end{align*}
