\section{MMSE (Minimum mean square error).}
\[ C(\theta,a) = \|\theta - a\|_2^2 \]
\[ \bar\theta = E[\theta \mid \mathcal{X}] \]
\[ E\{C(\theta,a) \mid \mathcal{X}\} = E\bigl[(\theta-a)^T(\theta-a) \mid \mathcal{X}\bigr] \]
\[ \frac{\partial}{\partial a}\, E\{(\theta-a)^T(\theta-a) \mid \mathcal{X}\} = 0 \]
\[ \frac{\partial}{\partial a}\, E\{\theta\theta^T - \theta \cdot a^T - a\theta^T + a a^T \mid \mathcal{X}\} = 0 \]
\[ 0 - \frac{\partial\, \bar\theta a^T}{\partial a} - \frac{\partial\, a\bar\theta^T}{\partial a} + \frac{\partial (a a^T)}{\partial a} = 0 \]
\[ -\bar\theta^T - \bar\theta + 2a^T = 0 \;\Rightarrow\; a = \bar\theta \]
\[ \hat\theta = \bar\theta \]
