\subsection*{Fourier dönüşümü}

\[ x(t) \;\longleftrightarrow\; X(f)=\int_{-\infty}^{\infty} x(t)\,e^{-j2\pi f t}\,dt \]
DSP $\searrow$
\[ x[n] \;\longleftrightarrow\; \underset{k=0,\dots,N-1}{X[k]} = \sum_{n=0}^{N-1} x[n]\,e^{-j\,{\color{kirmizikalem}\frac{2\pi}{N}k}\,n} \]

yeni notasyon
\[ \underset{u=0,\dots,{\color{kirmizikalem}N-1}}{F[u]} = \sum_{x=0}^{N-1} f[x]\,e^{-j\frac{2\pi}{{\color{kirmizikalem}N}}ux} \]

\begin{align*}
F[u,v] &= \sum_{x=0}^{M-1}\sum_{y=0}^{N-1} f(x,y)\,\underbrace{e^{-j\frac{2\pi}{M}ux}\,e^{-j\frac{2\pi}{N}vy}}_{e^{-j2\pi\left(\frac{ux}{M}+\frac{vy}{N}\right)}} \\
u &= 0,\dots,M-1 \\
v &= 0,\dots,N-1
\end{align*}

\[ f(x,y) = \overset{\color{kirmizikalem}?}{\dfrac{1}{MN}} \sum_{u=0}^{M-1}\sum_{v=0}^{N-1} F(u,v)\,e^{j\frac{2\pi}{M}ux}\,e^{j\frac{2\pi}{N}vy} \]

\[ f(x,y)\,\exp\!\left(2\pi j\left(\frac{u_0x}{M}+\frac{v_0y}{N}\right)\right) \;\longleftrightarrow\; F(u-u_0,\,v-v_0) \]

\[ u_0=\frac{M}{2}\ ,\quad v_0=\frac{N}{2} \]

\begin{align*}
e^{j2\pi\frac{M}{2}\frac{x}{{\color{kirmizikalem}M}}} f(x,y) &\longleftrightarrow \sum_{x=0}^{M-1}\left[e^{j2\pi\frac{M}{2}\frac{x}{{\color{kirmizikalem}M}}} f(x)\right] e^{-j\frac{2\pi}{M}ux} \\
&= \sum_{x=0}^{M-1} f(x)\,e^{-j\frac{2\pi}{{\color{kirmizikalem}M}}\left(\frac{u}{M}-\underbrace{\left(\frac{M}{2}\right)}_{u_0}\right)x}
\end{align*}

\begin{align*}
\sum_{x=0}^{M-1}\left[f(x)\,e^{j2\pi\frac{u_0x}{M}}\right]e^{-j\frac{2\pi}{M}ux} &= \sum_{x=0}^{M-1} f(x)\,e^{-j\frac{2\pi}{M}(u-u_0)x} \\
&= F(u-u_0)
\end{align*}

\begin{align*}
& u_0=\frac{M}{2} \\
& f(x) = e^{j2\pi\frac{\frac{M}{2}x}{M}} \\
& f(x)\left(\underbrace{e^{j\pi}}_{-1}\right)^{x} \\
& f(x)(-1)^x \longleftrightarrow F\!\left(u-\frac{M}{2}\right) \\
& f(x,y)(-1)^{x+y} \longleftrightarrow F\!\left(u-\frac{M}{2},\,v-\frac{N}{2}\right)
\end{align*}
