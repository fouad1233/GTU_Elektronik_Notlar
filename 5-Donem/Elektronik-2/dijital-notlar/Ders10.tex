\elcizimi[0.9]{sekiller/Ders10-s2-1.png}[2]

\begin{align*}
g_m &= \frac{I_E}{V_T} = \frac{1\,\mathrm{mA}}{25\,\mathrm{mA}\,\footnotemark} = 40\,\mathrm{mS}\\
r_e &= \frac{1}{g_m} \simeq 25\,\Omega\\
r_\pi &= 25\,\Omega \times 101 = 2{,}5\,\mathrm{k}\Omega\\
r_o &= \frac{V_A}{I_C} = \frac{100}{1} = 100\,\mathrm{k}\Omega
\end{align*}
\footnotetext{Claude notu: muhtemelen $25\,\mathrm{mV}$ olmalı.}
\begin{align*}
\frac{v_{out}}{v_{sig}} &= \frac{v_{out}}{v_\pi}\cdot\frac{v_\pi}{v_{sig}}\\
v_{out} &= -g_m v_\pi\,(r_o \parallel R_C \parallel R_L)\\
\frac{v_{out}}{v_\pi} &= -40\,(2{,}91)\\
\frac{v_{out}}{v_\pi} &= -116{,}5\\
v_\pi &= v_{sig}\,\frac{(R_B \parallel r_\pi)}{R_{sig} + (R_B \parallel r_\pi)}\\
&= \frac{2{,}43}{7{,}43}\,v_{sig}\\
\frac{v_\pi}{v_{sig}} &= \frac{2{,}43}{7{,}43} = 0{,}32\\
\frac{v_{out}}{v_{sig}} &= -116{,}5 \times 0{,}32 = -38{,}1\ [\mathrm{V/V}]\\
&= 31{,}6\ \mathrm{dB}
\end{align*}

\end{align*}
\begin{align*}

\tau_3 &= 12{,}4\,\mathrm{k}\Omega \times 1\,\mu\mathrm{F} = 12{,}4\,\mathrm{msec}\\
