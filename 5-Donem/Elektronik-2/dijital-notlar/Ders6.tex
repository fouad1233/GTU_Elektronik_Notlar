\end{ornek}
\elcizimi[0.55]{sekiller/Ders6-s3-1.png}[3]
\begin{align*}
& I_{D1} + I_{D2} = \qty{0,8}{\milli\ampere} \\
& \text{since } M_1 \equiv M_2 \\
& 2I_{D1} = 2I_{D2} = \qty{0,8}{\milli\ampere} \\
& I_{D1} = I_{D2} = I/2 = \qty{0,4}{\milli\ampere}
\end{align*}

\[ \qty{0,4}{\milli\ampere} = \frac{1}{2}\cdot 0{,}2\cdot 100\,(V_{ov})^2\footnotemark \]
\footnotetext{Çarpımın üzerine $0{,}1$ ve $10$ yazılmış (ara sonuçlar).}
