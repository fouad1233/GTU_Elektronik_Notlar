\section{Gradyan}

\[ \underbrace{\nabla f(x,y,z)}_{\textcolor{kirmizikalem}{\ast}\ =\ \mathrm{grad}\,f}=\frac{\partial f}{\partial x}\vec e_x+\frac{\partial f}{\partial y}\vec e_y+\frac{\partial f}{\partial y}\vec e_z\footnotemark\quad:\ \text{Gradyant vektöreldir} \]

\section{Divergans}
\[ \vec A=A_x\vec e_x+A_y\vec e_y+A_z\vec e_z \]
\[ \nabla\cdot\vec A=\mathrm{div}\,\vec A=\left(\frac{\partial}{\partial x}\vec e_x+\frac{\partial}{\partial y}\vec e_y+\frac{\partial}{\partial z}\vec e_z\right)\left(A_x\vec e_x+A_y\vec e_y+A_z\vec e_z\right) \]
\[ \mathrm{div}\,\vec A=\frac{\partial A_x}{\partial x}+\frac{\partial A_y}{\partial y}+\frac{\partial A_z}{\partial z} \]
\begin{center}
\blokdiyagram{vektör}{div}{skaler}
\end{center}
\[ \textcolor{kirmizikalem}{\begin{array}{ll}
\vec e_x\cdot\vec e_x=1 & \vec e_x\times\vec e_x=0 \\
\vec e_x\cdot\vec e_y=0 & \vec e_x\times\vec e_y=\vec e_z \\
 & \vec e_y\times\vec e_x=-\vec e_z \\
 & \vec e_y\times\vec e_z=\vec e_x \\
 & x\to y\to z\ \text{(döngüsel, ok z'den x'e geri döner)}
\end{array}} \]

\section{Rotasyonel}

\vec e_x & \vec e_y & \vec e_z \\

A_x & A_y & A_z
\end{vmatrix} \]
\[ =\left[\frac{\partial A_z}{\partial y}-\frac{\partial A_y}{\partial z}\right]\vec e_x+\left[\frac{\partial A_x}{\partial z}-\frac{\partial A_z}{\partial x}\right]\vec e_y+\left[\frac{\partial A_y}{\partial x}-\frac{\partial A_x}{\partial y}\right]\vec e_z \]
