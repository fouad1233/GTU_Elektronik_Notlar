\sayfa{2}
\sekilnotu[2]{Bu sayfa basılı materyal içeriyor (ör. ders slaytı); dijitalleştirilmedi. Basılı soru metni ve blok diyagram sayfa 1'dekiyle aynı; Q (25 p.) ifadesinin üstüne el yazısıyla 30 p. yazılmış.}

\noindent $x(t)=m(t)+\cos(2\pi f_c t)$\qquad $\cos^2(2\pi f_c t)=\tfrac12+\tfrac12\cos(2\pi\,2f_c t)$\qquad \okunamadi{$\ldots m(t)\cos(2\pi f_c t)+b$}

\noindent a)
\begin{align*}
z(t) &= 10a\,x(t)+b\,x^2(t)\\
 &= 10a\,m(t)+10a\cos(2\pi f_c t)+b\,m^2(t)+2b\,m(t)\cos(2\pi f_c t)\\
 &\quad +\tfrac{b}{2}+\tfrac{b}{2}\cos(2\pi\,2f_c t)
\end{align*}

\noindent i) $a=0\quad b=1 \;\Rightarrow\; z(t)=m^2(t)+2\,m(t)\cos(2\pi f_c t)+\tfrac12+\tfrac12\cos(2\pi\,2f_c t)$\\
\hspace*{2em}BPF at $f_c$ \qquad BW$=2B$ \qquad $y(t)=2\,m(t)\cos(2\pi f_c t)$

\noindent ii) $a=1\quad b=1$
\begin{align*}
z(t)&=10\,m(t)+\underline{10\cos(2\pi f_c t)}+m^2(t)+\underline{2\,m(t)\cos(2\pi f_c t)}\\
&\quad+\tfrac12+\tfrac12\cos(2\pi\,2f_c t)
\end{align*}
BPF at $f_c$ \qquad BW$=2B$ \qquad $y(t)=10\cos(2\pi f_c t)+2\,m(t)\cos(2\pi f_c t)$
\[
y(t)=10\left[1+\frac{2\,m(t)}{10}\right]\cos(2\pi f_c t)
\]

\noindent iii) $a=0\quad b=1 \;\Rightarrow$
\[
z(t)=m^2(t)+2\,m(t)\cos(2\pi f_c t)+\tfrac12+\tfrac12\cos(2\pi\,2f_c t)
\]
BPF at $f_c+\tfrac{B}{2}$ \qquad BW$=B$
\[
y(t)=2\left[m(t)\cos(2\pi f_c t)-m_h(t)\sin(2\pi f_c t)\right]
\]

\noindent b) $f_c=1\mathrm{k}\qquad m(t)=4\cos(2\pi\,100t)\qquad m_p=4\qquad m_n(t)=\cos(2\pi\,100t)$
\begin{align*}
y(t) &= 10\left[1+\frac{8}{10}\,\underbrace{m_n(t)}_{\cos(2\pi\,100t)}\right]\cos(2\pi\,1\mathrm{k}\,t)\qquad \mu=0{,}8\\
 &= 10\cos(2\pi\,1\mathrm{k}\,t)+8\,[\cos(2\pi\,100t)\cdot\cos(2\pi\,1\mathrm{k}\,t)]\\
 &= 10\cos(2\pi\,1\mathrm{k}\,t)+4\cos(2\pi\,1{,}1\mathrm{k}\,t)+4\cos(2\pi\,0{,}9\mathrm{k}\,t)
\end{align*}
\begin{align*}
Y(f) &= 5\,[\delta(f+1\mathrm{k})+\delta(f-1\mathrm{k})]\\
 &\quad +2\,[\delta(f+1{,}1\mathrm{k})+\delta(f-1{,}1\mathrm{k})+\delta(f+0{,}9\mathrm{k})+\delta(f-0{,}9\mathrm{k})]
\end{align*}
\begin{align*}
\eta &= \frac{\mu^2 P_{m_n}}{1+\mu^2 P_{m_n}}\qquad P_{m_n}=\tfrac12\quad\text{Ton modülasyonu}\\
 &= \frac{\mu^2}{2+\mu^2}\\
 &\cong \%24
\end{align*}
