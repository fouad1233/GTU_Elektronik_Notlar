\subsection{Mason's rule}
\[ G(s) = \frac{C(s)}{R(s)} = \frac{\sum_{k=1} T_k \Delta_k}{\Delta} \]


\sayfa{8}
\noindent $k$: number of forward paths

\noindent $T_k$: forward path gain for path \# $k$
\[ \Delta = 1 - \sum \text{loop gains} + \sum \begin{array}{c}\text{\small non touching loop gains}\\ \text{\small taken 2 at a time}\end{array} - \sum \begin{array}{c}\text{\small non touching loop gains}\\ \text{\small taken 3 at a time}\end{array} \]
\[ + \sum \begin{array}{c}\text{\small non touching loop gains}\\ \text{\small taken 4 at a time}\end{array} \ \cdots \]
\[ \Delta_k = \Delta - \sum \begin{array}{l}\text{\small loop gains that touch}\\ \text{\small the } k^{\text{th}} \text{ forward path}\end{array} \]

\[ \Delta = 1 + G_1G_2H_1 + G_2H_2 + G_3H_3 + (G_2G_3H_2H_3 + G_1G_2G_3H_1H_3) \]

\[ \Delta_k = \Delta - \sum \begin{array}{l}\text{\small loop gains that touch}\\ \text{\small the } k^{\text{th}} \text{ forward path}\end{array} \]
\noindent \hspace*{3em}(make the touching gain 0 in $\Delta$)
\[ \Delta_1 = 1 \qquad (H_1,H_2,H_3=0) \]
\[ \Delta_2 = 1 \]
\[ \frac{C(s)}{R(s)} = \frac{T_1\overset{1}{\Delta_1} + T_2\overset{1}{\Delta_2}}{\Delta} = \frac{G_1G_2G_3 + G_1G_3}{\Delta} \]
