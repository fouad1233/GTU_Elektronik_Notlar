\sayfa{7}
Show $h[n] = h[M-n]$, $M$ odd \hfill (tip 2)

$\Rightarrow$ get linear phase
\begin{align*}
M = 5 \qquad h[0] &= h[5]\\
h[1] &= h[4]\\
h[2] &= h[3]
\end{align*}
\begin{align*}
H(e^{j\omega}) &= \sum_{n=0}^{M} h[n]e^{-j\omega n}\\
&= \sum_{n=0}^{\frac{M-1}{2}} h[n]e^{-j\omega n} + \sum_{n=\frac{M+1}{2}}^{M} h[n]e^{-j\omega n}\\
&= \sum_{n=0}^{\frac{M-1}{2}} h[n]e^{-j\omega n} + \sum_{k=0}^{\frac{M-1}{2}} h[M-k]e^{-j\omega(M-k)}\\
&= \sum_{n=0}^{\frac{M-1}{2}} h[n]e^{-j\omega n} + \left(\sum_{n=0}^{\frac{M-1}{2}} h[n]e^{+j\omega n}\right)e^{-j\omega M}\\
&= \sum_{n=0}^{\frac{M-1}{2}} h[n]\left[e^{-j\omega n} + e^{j\omega n}e^{-j\omega M}\right]\\
&= \sum_{n=0}^{\frac{M-1}{2}} h[n]\Big[e^{-j\omega n}e^{-j\omega\frac{M}{2}}e^{j\omega\frac{M}{2}}\\
&\qquad\qquad\quad + e^{j\omega n}e^{-j\omega\frac{M}{2}}e^{-j\omega\frac{M}{2}}\Big]\\
&= e^{-j\omega\frac{M}{2}}\sum_{n=0}^{\frac{M-1}{2}} h[n]\left[e^{-j\omega(n-\frac{M}{2})} + e^{j\omega(n-\frac{M}{2})}\right]\\
H(e^{j\omega}) &= e^{-j\omega\frac{M}{2}}\; 2\sum_{n=0}^{\frac{M-1}{2}} h[n]\cos\left(\omega n - \omega\frac{M}{2}\right)
\end{align*}
\[ |H(e^{j\omega})| = \Big|\,2\underbrace{\sum_{n=0}^{\frac{M-1}{2}} h[n]\cos\left(\omega n - \omega\frac{M}{2}\right)}_{\downarrow}\Big| \]
\[ \angle H(e^{j\omega}) = \begin{cases} -\omega\frac{M}{2} & > 0\\[2pt] -\omega\frac{M}{2} + \pi & < 0 \end{cases} \]
