\section{Likelihood Ratio Test and Maximum a Posteriori\textsuperscript{\textcolor{kirmizikalem}{\itshape probability}} (MAP) Criterion}
\noindent Muhtemellik oranı testi ve maksimum sonsal olasılık kriteri

\[ P(s_1/z) \overset{H_1}{\underset{H_2}{\gtrless}} P(s_2/z) \qquad (H_i:\ \text{Hipotez}) \]
\[ P(s_i/z) = \frac{f(z/s_i)\,P(s_i)}{f(z)} \qquad i=1,2 \]
\[ f(z) = \sum_{i=1}^{2} f(z/s_i)\,P(s_i) = f(z/s_1)\,P(s_1) + f(z/s_2)\,P(s_2) \]
\[ f(z/s_1)\,P(s_1) \overset{H_1}{\underset{H_2}{\gtrless}} f(z/s_2)\,P(s_2) \]
\[ \Lambda(z) = \frac{f(z/s_1)}{f(z/s_2)} \overset{H_1}{\underset{H_2}{\gtrless}} \frac{P(s_2)}{P(s_1)} \qquad \text{likelihood ratio test} \]
\[ \text{Eğer } P(s_1)=P(s_2)=\tfrac{1}{2} \ \Rightarrow\ \Lambda(z) = \frac{f(z/s_1)}{f(z/s_2)} \overset{H_1}{\underset{H_2}{\gtrless}} 1 \]
\noindent Maximum likelihood criterion\\ (En muhtemeli bulma kriteri)

\sayfa{8}
\[
\Lambda(z) = \exp\Biggl\{\underbrace{-\frac{z^2+a_1^2-2za_1}{2\sigma_0^2} + \frac{z^2+a_2^2-2za_2}{2\sigma_0^2}}_{\displaystyle\frac{a_2^2-a_1^2}{2\sigma_0^2} + \frac{2z(a_1-a_2)}{2\sigma_0^2}}\Biggr\} \overset{H_1}{\underset{H_2}{\gtrless}} \frac{P(s_2)}{P(s_1)}
\]
\[ L(z) = \ln\Lambda(z) = \frac{a_2^2-a_1^2}{2\sigma_0^2} + \frac{z(a_1-a_2)}{\sigma_0^2} \overset{H_1}{\underset{H_2}{\gtrless}} \ln\frac{P(s_2)}{P(s_1)} \]
\[ \frac{z(a_1-a_2)}{\sigma_0^2} \overset{H_1}{\underset{H_2}{\gtrless}} \ln\frac{P(s_2)}{P(s_1)} + \frac{a_1^2-a_2^2}{2\sigma_0^2} \]
\[ z \overset{H_1}{\underset{H_2}{\gtrless}} \frac{\sigma_0^2}{a_1-a_2}\,\ln\frac{P(s_2)}{P(s_1)} + \frac{\sigma_0^2}{a_1-a_2}\,\frac{a_1^2-a_2^2}{2\sigma_0^2} \]
\noindent {\small\textcolor{turuncukalem}{$a_1+a_2$ kalır}}
\[ \kutula{z \overset{H_1}{\underset{H_2}{\gtrless}} \frac{\sigma_0^2}{a_1-a_2}\,\ln\frac{P(s_2)}{P(s_1)} + \frac{a_1+a_2}{2}}\ \textcolor{kirmizikalem}{=\gamma_0} \]

\[ \text{Eğer } P(s_1)=P(s_2)=\tfrac{1}{2} \quad\Longrightarrow\quad \gamma_0 = \frac{a_1+a_2}{2} \]

\[ P(h/s_1) = \int_{-\infty}^{\gamma_0} f(z/s_1)\,dz = \frac{1}{\sqrt{2\pi}\,\sigma_0}\int_{-\infty}^{\gamma_0} e^{-\frac{(z-a_1)^2}{2\sigma_0^2}}\,dz \]
\begin{align*}
&= 1 - \frac{1}{\sqrt{2\pi}\,\sigma_0}\int_{\gamma_0}^{\infty} e^{-\frac{(z-a_1)^2}{2\sigma_0^2}}\,dz\\
&= 1 - \underbrace{\frac{1}{\sqrt{2\pi}\,\sigma_0}\int_{\frac{\gamma_0-a_1}{\sigma_0}}^{\infty} e^{-t^2/2}\,\sigma_0\,dt}_{Q\left(\frac{\gamma_0-a_1}{\sigma_0}\right)}
\end{align*}
\begin{align*}
\frac{z-a_1}{\sigma_0} = t \qquad\qquad & z=\gamma_0 \ \Rightarrow\ t = \frac{\gamma_0-a_1}{\sigma_0}\\
& z=\infty \ \rightarrow\ t=\infty
\end{align*}
\begin{align*}
\Rightarrow\ z &= \sigma_0 t + a_1\\
dz &= \sigma_0\,dt
\end{align*}
\[ \Rightarrow\ \kutula{P(h/s_1) = 1 - Q\!\left(\frac{\gamma_0-a_1}{\sigma_0}\right)} \]

\[ P(h/s_2) = \int_{\gamma_0}^{\infty} f(z/s_2)\,dz \]
\begin{align*}
&= \frac{1}{\sqrt{2\pi}\,\sigma_0}\int_{\gamma_0}^{\infty} e^{-\frac{(z-a_2)^2}{2\sigma_0^2}}\,dz\\
&= \underbrace{\frac{1}{\sqrt{2\pi}\,\sigma_0}\int_{\frac{\gamma_0-a_2}{\sigma_0}}^{\infty} e^{-t^2/2}\,\sigma_0\,dt}_{\textcolor{turuncukalem}{Q\left(\frac{\gamma_0-a_2}{\sigma_0}\right)}}
\end{align*}

\sayfa{9}
\[ P(h/s_2) = Q\!\left(\frac{\gamma_0-a_2}{\sigma_0}\right) \]
\[ P_b = P(h/s_1)\cdot P(s_1) + P(h/s_2)\cdot P(s_2) \]

\[ Q(-x) = 1 - Q(x) \]
