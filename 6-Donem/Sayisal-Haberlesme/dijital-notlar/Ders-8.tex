\section{Karşılayıcı Filtre (Matched Filter - MF)}
\begin{itemize}
\item MF provides max SNR at its output for a given transmitted waveform
\end{itemize}

\elcizimi[0.8]{sekiller/Ders-8-s3-1.png}[3]

\[ \mathrm{SNR}=\left(\frac{S}{N}\right)_{t=T}=\frac{a^2}{\underset{N_0/2}{\sigma_0^2}}=\frac{2E_s}{N_0}\qquad \text{eğer } h(t)=s(T-t),\ 0\le t\le T \]
\[ E_s=\int_0^T |s(t)|^2\,dt \]

\sayfa{4}
\[ z(t)=r(t)\circledast h(t)=\underbrace{s(t)\circledast h(t)}_{a(t)}+\underbrace{n(t)\circledast h(t)}_{n_0(t)} \]

\[ z(T)=a(T)+n_0(T) \]
\[ \begin{array}{l} z=a+n_0 \\ \qquad\ \hookrightarrow\ N(0,\sigma_0^2) \end{array} \]

\[ \mathrm{SNR}=\left(\frac{S}{N}\right)_{T}=\frac{a^2}{\sigma_0^2} \]

\[ a(t)=\iFou{A(f)}=\int_{-\infty}^{\infty}S(f)H(f)e^{j2\pi fT}\,df \]
\[ \rightarrow a=a(T) \]

\elcizimi[0.85]{sekiller/Ders-8-s4-1.png}[4]

\[ P_{n_0}=\sigma_0^2=\int_{-\infty}^{\infty}S_{n_0}(f)\,df=\frac{N_0}{2}\int_{-\infty}^{\infty}|H(f)|^2\,df \]

\[ \mathrm{SNR}=\frac{a^2}{\sigma_0^2}=\frac{\left|\int_{-\infty}^{+\infty}S(f)H(f)e^{j2\pi fT}\,df\right|^2}{\frac{N_0}{2}\int_{-\infty}^{\infty}|H(f)|^2\,df} \]

Cauchy-Schwarz eşitsizliği
\[ \left|\int_{-\infty}^{\infty}f_1(x)f_2(x)\,dx\right|^2\le\int_{-\infty}^{\infty}|f_1(x)|^2dx\int_{-\infty}^{\infty}|f_2(x)|^2dx \]
\[ \hookrightarrow\ f_1(x)=k\cdot f_2^{*}(x)\qquad\text{\okunamadi{şartı}} \]

\[ \begin{aligned} f_1(x)&=H(f)\\ f_2(x)&=S(f)e^{j2\pi fT} \end{aligned} \]
\[ \Rightarrow\ \mathrm{SNR}\le\frac{\int_{-\infty}^{\infty}|H(f)|^2df\int_{-\infty}^{\infty}|S(f)|^2df}{\frac{N_0}{2}\int_{-\infty}^{\infty}|H(f)|^2df} \]

\[ \mathrm{SNR}\le\frac{2}{N_0}\underbrace{\int_{-\infty}^{\infty}|S(f)|^2\,df}_{E_s}\ \Rightarrow\ \mathrm{SNR}\le\frac{2E_s}{N_0} \]

SNR'ın maksimum değeri \okunamadi{budur}

\sayfa{5}

\[ \begin{aligned} f_1(x)&=k\,f_2^{*}(x) \\ H(f)&=k\,S^{*}(f)\,e^{-j2\pi fT} \end{aligned} \]
\[ \downarrow \]
\[ h(t)=k\int_{-\infty}^{\infty}\underbrace{S^{*}(f)\,e^{-j2\pi fT}}\,e^{j2\pi ft}\,df \]
\[ =k\left[\underbrace{\int_{-\infty}^{\infty}S(f)\,e^{j2\pi f\underbrace{(T-t)}_{t'}}\,df}_{s(t')}\right]^{*} \]
\[ h(t)=k\cdot s^{*}(t')=k\cdot s^{*}(T-t)\ \rightarrow\ \kutula{h(t)=s^{*}(T-t)} \]
\[ h(t)=s(T-t)\ \rightarrow\ \mathrm{SNR}=\frac{2E_s}{N_0} \]

\[ \left(\frac{S}{N}\right)_T=\frac{(a_1-a_2)^2}{\sigma_0^2}=\frac{2\overbrace{E_d}^{\substack{\text{\scriptsize Fark işaretin enerjisi}\\ \text{\scriptsize (Difference signal)}}}}{N_0} \]
$\leftarrow$ $h(t)$ \ $s_1(t)-s_2(t)$'yi karşılıyor

\[ E_d=\int_0^T\bigl[s_1(t)-s_2(t)\bigr]^2dt=\underbrace{\int_0^T s_1^2(t)\,dt}_{\substack{E_b:\ \text{bit başına}\\ \text{energy per bit}}}+\underbrace{\int_0^T s_2^2(t)\,dt}_{E_b}-2\underbrace{\int_0^T s_1(t)s_2(t)\,dt}_{\rho\cdot E_b} \]

\[ \rho=\frac{1}{E_b}\int_0^T s_1(t)s_2(t)\,dt \]
$\downarrow$ \underline{karşı-ilişki katsayısı} \\
(Normalized cross correlation coefficient) \\
$-1\le\rho\le 1$

\textit{[Ok, ``(Normalized cross correlation coefficient)'' ifadesinden $\rho\cdot E_b$ ifadesine gidiyor.]}

\[ E_d=2E_b-2\rho E_b=2E_b(1-\rho) \]

\[ \begin{aligned} P_b&=Q\left(\frac{a_1-a_2}{2\sigma_0}\right)=Q\left(\sqrt{\frac{2E_d}{4N_0}}\right) \\ &=Q\left(\sqrt{\frac{E_d}{2N_0}}\right)=Q\left(\sqrt{\frac{E_b}{N_0}(1-\rho)}\right) \end{aligned} \]
