\begin{ornek}[5.1]
  $L_{min} = 0{,}4\ \mu\mathrm{m}$ \hfill $t_{ox} = 8\ \mathrm{nm}$ \hspace{2cm} \\
  $\mu_n = 450\ \mathrm{cm^2/V\cdot s}$ \hfill $V_{TH} = 0{,}7\ \mathrm{V}$ \hspace{2cm}

  a) Find $C_{ox}$ and $k_n'$

  b) $\dfrac{W}{L} = \dfrac{8\ \mu\mathrm{m}}{0{,}8\ \mu\mathrm{m}}$ $\rightarrow$ $V_{OV},\ V_{GS},\ V_{DS\,\mathrm{min}}$ $\rightarrow$ needed to operate the transistor in the sat. region $\rightarrow$ $I_D = 100\ \mu\mathrm{A}$

  c) (b) deki transistör için $V_{OV}$ ve $V_{GS}$ gerilimlerini hesaplayınız \\
  $\hookrightarrow$ $1000\ \Omega$ Resistor
\end{ornek}
\begin{cozum}
  a) $C_{ox} = \dfrac{\varepsilon_{ox}}{t_{ox}}$ \quad $\rightarrow$ \quad $\varepsilon_{ox} = \varepsilon_r \cdot \varepsilon_0$
  \quad ($\varepsilon_r$: 3,9 for SiO$_2$; $\varepsilon_0$: $8{,}85\cdot10^{-12}\ \mathrm{F/m}$)
  \[ = \frac{3{,}45\cdot10^{-11}}{8\times10^{-9}} = 4{,}32\times10^{-3}\ \mathrm{F/m^2} = 4{,}32\ \mathrm{fF/\mu m^2} \]
  \begin{align*}
    k_n' = \mu_n C_{ox} &= (450\ \mathrm{cm^2/V\cdot s})(4{,}32\ \mathrm{fF/\mu m^2}) \\
    &= 194\times10^{-6}\ \mathrm{F/V\cdot s} \qquad \rightarrow\ C = \frac{Q}{V} \Rightarrow \frac{\mathrm{F}}{\mathrm{V\cdot s}} \\
    &= 194\ \mu\mathrm{A/V^2}
  \end{align*}
  \hfill $\Rightarrow\ \dfrac{Q}{V^2\cdot\mathrm{s}} = \dfrac{I\cdot\mathrm{s}}{V^2\cdot\mathrm{s}} = \dfrac{I}{V^2}$

  \hfill $I = \dfrac{Q}{\mathrm{s}} \;\rightarrow\; \dfrac{dq}{dt} = i$

  b) Sat
  \[ I_D = \frac{1}{2}k_n'(V_{OV})^2 \]
  \[ 100 = \frac{1}{2}\times194\times\frac{8}{0{,}8}\,(V_{OV})^2 \]
  \[ V_{OV} = 0{,}32\ \mathrm{V} \]
  \[ V_{GS} = \underset{0{,}7}{V_{TH}} + \underset{0{,}32}{V_{OV}} = 1{,}02\ \mathrm{V} \]
\end{cozum}
