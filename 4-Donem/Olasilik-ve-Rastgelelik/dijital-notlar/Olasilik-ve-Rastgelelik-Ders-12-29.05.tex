\sayfa{8}
\[
\textcircled{3}\text{\footnotemark}\quad f(x,y)=\begin{cases}6e^{-2x-3y}, & 0<x,y<\infty\\[2mm] 0, & \text{diğer}\end{cases}
\]
\footnotetext{Claude notu: orijinal dosyadaki basılı ödev sorusunda bu, 2. sorudur.}
\[
\textbf{a)}\quad \mathrm{Cov}(X,Y)=E[XY]-E[X]\,E[Y]
\]
\begin{align*}
E[X]&=\int_0^\infty\!\!\int_0^\infty x\,6e^{-2x-3y}\,dx\,dy\\
&=6\int_0^\infty e^{-3y}\,dy\int_0^\infty xe^{-2x}\,dx\qquad
\left(\begin{array}{ll}u=x & du=dx\\ e^{-2x}dx=dv & v=-\frac12\,e^{-2x}\end{array}\right.\\
&=6\Bigl(-\frac13\Bigr)e^{-3y}\Big|_0^\infty\cdot\Bigl[-\frac{xe^{-2x}}{2}\Big|_0^\infty+\int_0^\infty\frac{e^{-2x}}{2}\,dx\\
&=2\cdot\Bigl(-\frac14\Bigr)e^{-2x}\Big|_0^\infty=-\frac12\,(0-1)=\frac12
\end{align*}

\sayfa{9}
\begin{align*}
E[Y]&=\int_0^\infty\!\!\int_0^\infty y\,6e^{-2x}e^{-3y}\,dx\,dy\\
&=6\int_0^\infty e^{-2x}\,dx\int_0^\infty ye^{-3y}\,dy\qquad
\left(\begin{array}{ll}u=y & du=dy\\ e^{-3y}dy=dv & -\frac13\,e^{-3y}=v\end{array}\right.\\
&=6\Bigl(-\frac12\Bigr)e^{-2x}\Big|_0^\infty\Bigl[-\frac{ye^{-3y}}{3}\Big|_0^\infty+\int_0^\infty\frac{e^{-3y}}{3}\,dy\Bigr]\\
&=3\Bigl(-\frac19\Bigr)e^{-3y}\Big|_0^\infty=\frac13
\end{align*}

\sayfa{12}
\[
\okunamadi{3}\quad E[X|Y=y]=\int_{-\infty}^{\infty}x\,f_{X|Y}(x|y)\,dx
\]
\[
f_{X|Y}(x|y)=\frac{f_{X,Y}(x,y)}{f_Y(y)}
\]
\begin{align*}
f_Y(y)&=\int_{-\infty}^{\infty}f_{X,Y}(x,y)\,dx=\int_0^\infty e^{-x/y}e^{-y}\,dx\\
&=-y\,e^{-x/y}e^{-y}\Big|_0^\infty=-y\,e^{-y}\,(0-1)=y\,e^{-y}
\end{align*}
\[
f_{X|Y}(x|y)=\frac{e^{-x/y}e^{-y}}{y\,e^{-y}}=\frac1y\,e^{-x/y}
\]

\sayfa{13}
\begin{align*}
E[X|Y=y]&=\int_0^\infty\frac{x}{y}\,e^{-x/y}\,dx=\qquad
\left(\begin{array}{ll}x=u & dx=du\\ \dfrac1y\,e^{-x/y}dx=dv & -e^{-x/y}=v\end{array}\right.\\
&=-xe^{-x/y}\Big|_0^\infty+\int_0^\infty e^{-x/y}\,dx=-y\,e^{-x/y}\Big|_0^\infty=y
\end{align*}
\[
\kutula{E[X|Y=y]=y}
\]
