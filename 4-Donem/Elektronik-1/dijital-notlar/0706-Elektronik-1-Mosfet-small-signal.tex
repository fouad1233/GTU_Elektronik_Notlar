\section{Common source Amplifier}
\begin{itemize}[label=--]
  \item Voltage Amp \quad (High)
  \item $R_{in}=\infty$ \quad $\}$ Inf
  \item $R_{out}=r_o\,/\!/\,R_D\,/\!/\,R_L$ \quad $\Big\}$ High\\
        $R_{out}$ (intrinsic) $=r_o$
\end{itemize}

\sekilnotu[1]{Ortak kaynaklı yükselteç devresi: $V_{sig}$ kaynağı -- $R_{sig}$ -- kuplaj kondansatörü -- gate; gate'te $R_{G1}$ ($V_{DD}$'ye) ve $R_{G2}$ (toprağa) gerilim bölücüsü; drain $R_D$ üzerinden $V_{DD}$'ye bağlı ve $C_\infty$ ile $R_L$ yüküne kuplajlı; source'ta $R_S$ ve ona paralel $C_\infty$ (bypass) toprağa.}

$\Rightarrow$
\sekilnotu[1]{Küçük işaret eşdeğeri: $V_{sig}$ -- $R_{sig}$ -- $G$; $G$ ile $S$ arası açık uçlar ($+\ v_{GS}\ -$); $S$ toprağa; $D$ ile toprak arasında $g_mV_{GS}$ bağımlı akım kaynağı (aşağı yönlü), $r_o$, $R_D$ ve $R_L$ paralel; çıkış $v_{out}$ drain'den. Girişe ok: $R_{in}=R_{G1}/\!/R_{G2}$; $R_D$'ye ok: $R_{out}$.}
\[
  R_{in}=R_{G1}\,/\!/\,R_{G2} \qquad\qquad R_{out}=r_o\,/\!/\,R_D

\section{Common-Drain Amplifier \textcolor{murekkep}{(Source follower)}}
\sekilnotu[4]{Ortak drain'li yükselteç: $V_{sig}$ -- $R_{sig}$ -- gate; gate'te $R_{G1}$ ($V_{DD}$'ye) ve $R_{G2}$ (toprağa); drain $V_{DD}$'den aşağı yönlü bir akım kaynağı ile besleniyor; source $R_S$ ile toprağa ve $\infty$ kuplaj kondansatörü ile $R_L$ yüküne bağlı, çıkış $V_{out}$.}

\sekilnotu[4]{Küçük işaret eşdeğeri: $V_{sig}$ -- $R_{sig}$ -- $R_G$ (toprağa; $R_{in}=R_G$ oku) -- $G$ ($+$) ve $S$ ($-$) açık uçları arasında $V_{gs}$; kırmızı kutu içinde $g_mV_{gs}$ bağımlı akım kaynağı (aşağı yönlü) ve paralel $r_o$, kutunun üstü (drain) toprağa; kutunun altı (source) çıkış düğümüne iniyor; çıkış düğümünde $R_S$ ve $R_L$ toprağa, $V_{out}$; $r_o$ bir okla çıkış düğümü ile toprak arasına taşınmış olarak tekrar çizilmiş. $R_{out}\cong 1/g_m$.}
\[
  R_{in}=R_G \qquad\qquad R_{out}\cong 1/g_m
\]
\begin{align*}
  & V_{out}=g_mV_{gs}\,(R_S\,/\!/\,R_L\,/\!/\,r_o) \\
  & R_{sig}=0 \\
  & V_{gs}=V_{sig}-g_mV_{gs}\,(r_o\,/\!/\,R_L\,/\!/\,R_S) \\
  & V_{sig}=V_{gs}\,\bigl(1+g_m\,(r_o\,/\!/\,R_L\,/\!/\,R_S)\bigr)
\end{align*}
\[
  \Rightarrow\ A_v=\frac{g_mV_{gs}\,(R_L')}{1+g_mV_{gs}\,(R_L')}
  \ <1 \qquad\text{\footnotemark}
\]
\footnotetext{Claude notu: $V_{gs}$'ler sadeleşince muhtemelen $A_v=\dfrac{g_mR_L'}{1+g_mR_L'}$ olmalı.}
