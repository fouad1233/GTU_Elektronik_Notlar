  \frac{v_{out}}{v_{in}} &= \frac{-g_m\, (R_C \parallel R_L)}{r_e \left( 1 + \dfrac{R_E}{r_e} \right)}\footnotemark \\[2mm]
  &\cong -\frac{(R_C \parallel R_L)}{r_e + R_e} \\[2mm]
  A_v &\cong -\frac{R_C \parallel R_L}{R_E + r_e}
\end{align*}
