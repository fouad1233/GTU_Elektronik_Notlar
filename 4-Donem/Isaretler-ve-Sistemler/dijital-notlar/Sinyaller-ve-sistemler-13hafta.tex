\section{Ters Laplace Transform:}
\begin{minipage}{\linewidth}
\[ X(s)=X(\sigma+j\omega)=\int_{-\infty}^{\infty} x(t)\,e^{-\sigma t}\,e^{-st}\,dt=\Fou{x(t)\,e^{-\sigma t}}\footnote{Claude notu: muhtemelen $e^{-j\omega t}$ olmalı.} \]
\end{minipage}
\[ x(t)\,e^{-\sigma t}=\iFou{X(\sigma+j\omega)}=\frac{1}{2\pi}\int_{-\infty}^{\infty} X(\sigma+j\omega)\,e^{j\omega t}\,d\omega \]
\[ x(t)=\frac{1}{2\pi}\int_{-\infty}^{\infty} X(\sigma+j\omega)\,e^{(\sigma+j\omega)t}\,d\omega \]
\[ s=\sigma+j\omega \ \to\ ds=j\,d\omega \]
\[ x(t)=\frac{1}{2\pi j}\int_{\sigma-j\infty}^{\sigma+j\infty} X(s)\,e^{st}\,ds \]
\[ \to\ X(s)=\sum_{i=1}^{m}\frac{A_i}{s+a_i}\ \longrightarrow\
\begin{aligned}
  & x(t)=A_i\,e^{-a_i t}u(t)\ :\ \text{sağ yanlı},\\
  \hookrightarrow\ & x(t)=-A_i\,e^{-a_i t}u(-t)\ :\ \text{sol yanlı}
\end{aligned} \]

\sayfa{2}\nopagebreak
\section{Laplace Transformun özellikleri}
\[ x(t)\ \overset{\mathcal{L}}{\longleftrightarrow}\ X(s),\ \ \mathrm{ROC}:\ R \]

\begin{enumerate}[label=\arabic*), leftmargin=2em, itemsep=5pt]
  \item Lineerlik: $\alpha x_1(t)+\beta x_2(t)\donusum \alpha X_1(s)+\beta X_2(s)$, \ ROC: $R_1\cap R_2$
  \item Zamanda öteleme: $x(t-t_0)\donusum e^{-st_0}X(s)$, \ ROC: $R$
  \item S-domaininde öteleme: $e^{s_0 t}x(t)\donusum X(s-s_0)$, \ ROC: $R+\mathrm{Re}\{s_0\}$\\
    $s_0=j\omega_0\ \Rightarrow\ e^{j\omega_0 t}x(t)\donusum X(s-j\omega_0)$, \ ROC: $R$
  \item Zamanda ölçeklendirme: $x(at)=\dfrac{1}{|a|}X(s/a)$, \ ROC: $R/a$\\
    $a=-1\ \to\ x(-t)\donusum X(-s)$, \ ROC: $-R$
  \item Eşlenik alma: $x^{*}(t)\donusum X^{*}(s^{*})$, \ ROC: $R$
  \item Türev: $\dfrac{dx(t)}{dt}\donusum sX(s)$, \ ROC: $R$'yi içerir\\
    $-t\,x(t)\donusum\dfrac{dX(s)}{ds}$, \ ROC: $R$
  \item İntegral: $\displaystyle\int_{-\infty}^{t}x(\tau)\,d\tau\donusum\frac{1}{s}X(s)$, \ ROC: $R\cap\{\mathrm{Re}\{s\}>0\}$
  \item Konvolüsyon: $x_1(t)*x_2(t)\donusum X_1(s)X_2(s)$, \ ROC: $R_1\cap R_2$
\end{enumerate}

\section{z transformu}
\[ X(z)=\sum_{n=-\infty}^{\infty}x[n]\,z^{-n} \]
\[ x[n]\ \overset{z}{\longleftrightarrow}\ X(z) \]
\[ X(z)=\mathcal{Z}\{x[n]\} \]
\[ \Rightarrow\ z=re^{j\omega}\qquad
\begin{aligned}
  \longrightarrow\ & r:\ \text{magnitude}\\
  \hookrightarrow\ & \omega:\ \text{angle}
\end{aligned} \]
\begin{align*}
  X(z)=X(re^{j\omega})=\sum_{n=-\infty}^{\infty}x[n]\,(re^{j\omega})^{-n}
    &=\sum_{n=-\infty}^{\infty}x[n]\,r^{-n}e^{-j\omega n}\\
    &=\Fou{x[n]\,r^{-n}}
\end{align*}
\[ \Rightarrow\ z=e^{j\omega}\ (r=1) \]
\[ X(z)=X(e^{j\omega})=\sum_{n=-\infty}^{\infty}x[n]\,e^{-j\omega n}=\Fou{x[n]}=X(z)\Big|_{z=e^{j\omega}} \]

\begin{enumerate}[label=\arabic*), leftmargin=2em, itemsep=6pt]
  \item $X(z)$'nin ROC'u z düzleminde
  \item ROC, herhangi bir kutup içermez
  \item $x[n]$ sonlu süreli ise, ROC, olası $z=0$ ve/veya $z=\infty$ değerleri dışında bütün z-düzlemidir.

\begin{ornek}
  $x[n]=\delta[n]$
\end{ornek}
\[ X(z)=\sum_{n=-\infty}^{\infty}x[n]\,z^{-n}=\sum_{n=-\infty}^{\infty}\delta[n]\,z^{-n}=\underset{\mathclap{\begin{array}{c}\uparrow\\[-1mm] \delta[n]=1,\ n=0\end{array}}}{1}, \]
